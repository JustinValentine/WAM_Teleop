# The CANopen library's own hardware test programs (libbarrett-canopen/tests),
# built against the library the leader uses (/opt/barrett-canopen in wam-ros:4c).
cmake_minimum_required(VERSION 3.5)
project(lbc_tests CXX)
set(CMAKE_BUILD_TYPE RelWithDebInfo)
set(P /opt/barrett-canopen)
include(${P}/share/barrett/barrett-config.cmake)
list(REMOVE_ITEM BARRETT_LIBRARIES barrett)
set(SRC ${CMAKE_CURRENT_SOURCE_DIR}/../../libbarrett-canopen/tests)
foreach(t lbc_torque_sign lbc_position_watch)
  add_executable(${t} ${SRC}/${t}.cpp)
  target_include_directories(${t} BEFORE PRIVATE ${BARRETT_INCLUDE_DIRS} ${SRC})
  target_compile_options(${t} PRIVATE ${BARRETT_DEFINITIONS})
  target_link_libraries(${t} ${P}/lib/libbarrett.so ${BARRETT_LIBRARIES})
  set_target_properties(${t} PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "${P}/lib;/usr/local/lib")
endforeach()
